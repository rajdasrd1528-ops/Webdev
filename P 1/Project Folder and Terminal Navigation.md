# P1 – Project Folder Creation and Terminal Navigation

# 1. Check current directory
pwd

# 2. List files and folders
ls

# 3. Create project folder
mkdir raj-webdev-lab

# 4. Open project folder
cd raj-webdev-lab

# 5. Create subfolders
mkdir css images

# 6. Create HTML and CSS files
touch index.html
touch style.css

# 7. View all files
ls -la

# 8. Go back to previous directory
cd ..

# 9. Open project folder again
cd raj-webdev-lab
